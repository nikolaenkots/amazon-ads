# Расписание второй установки. Пути под VM margoads
# (/home/nikolaenko_ts/margoads); для другой машины поправьте путь.
# Ставится так:  crontab ~/margoads/deploy/crontab.txt
#
# Время указано по Варшаве: CRON_TZ снимает вопрос перехода на зимнее время,
# иначе запуск пришлось бы дважды в год сдвигать вручную. Проверить, что cron
# понял часовой пояс, можно по логу первого запуска.
CRON_TZ=Europe/Warsaw

# 10:00 — синхронизация структуры кампаний. Идёт первой, чтобы статистика
# легла на актуальный список кампаний и групп.
0 10 * * * cd $HOME/margoads && venv/bin/python scripts/auto_sync_campaigns.py >> $HOME/margoads/logs/sync.log 2>&1

# 11:00 — сбор статистики за последние 14 дней по всем гео и типам отчётов.
# Час разрыва: синхронизация 14 профилей занимает заметное время.
0 11 * * * cd $HOME/margoads && venv/bin/python scripts/auto_collect.py >> $HOME/margoads/logs/collect.log 2>&1
